'Entries: 1/999999 (Not Modified Often)'

The structure for the database is described below

-- Years table to define academic years
CREATE TABLE years (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL UNIQUE,
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
-- Indexes for years table
CREATE INDEX idx_years_name ON years(name);
CREATE INDEX idx_years_sort_order ON years(sort_order);
CREATE INDEX idx_years_is_active ON years(is_active);

-- Subjects table to define available subjects
CREATE TABLE subjects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL UNIQUE,
    code VARCHAR(10),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
-- Indexes for subjects table
CREATE INDEX idx_subjects_name ON subjects(name);
CREATE INDEX idx_subjects_code ON subjects(code);
CREATE INDEX idx_subjects_is_active ON subjects(is_active);

-- Users table with year_id reference
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255), -- Nullable for OAuth users
    email_verified BOOLEAN DEFAULT FALSE,
    role VARCHAR(50) DEFAULT 'user',
    year_id UUID,
    profile_picture_url VARCHAR(255) DEFAULT 'default',
    total_xp INTEGER DEFAULT 0 NOT NULL,
    current_streak INTEGER DEFAULT 0 NOT NULL,
    longest_streak INTEGER DEFAULT 0 NOT NULL,
    last_activity_date DATE,
    streak_freezes_remaining INTEGER DEFAULT 14 NOT NULL,      -- Freezes available to use (0-14)
    streak_freezes_used_this_year INTEGER DEFAULT 0 NOT NULL,  -- Freezes consumed this calendar year
    freeze_reset_year INTEGER DEFAULT EXTRACT(YEAR FROM CURRENT_DATE) NOT NULL,  -- Year when freezes were last reset
    has_completed_tutorial BOOLEAN DEFAULT FALSE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (year_id) REFERENCES years(id) ON DELETE SET NULL
);
-- Indexes for users table
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_created_at ON users(created_at);
CREATE INDEX idx_users_year_id ON users(year_id);
CREATE INDEX idx_users_total_xp ON users(total_xp DESC);
CREATE INDEX idx_users_has_completed_tutorial ON users(has_completed_tutorial);
CREATE INDEX idx_users_freeze_year ON users(freeze_reset_year);

-- OAuth providers table for third-party authentication
CREATE TABLE oauth_providers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    provider VARCHAR(50) NOT NULL,
    provider_user_id VARCHAR(255) NOT NULL,
    email VARCHAR(255),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    last_used_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE(provider, provider_user_id)
);
-- Indexes for oauth_providers table
CREATE INDEX idx_oauth_provider_user ON oauth_providers(provider, provider_user_id);
CREATE INDEX idx_oauth_user_id ON oauth_providers(user_id);
CREATE INDEX idx_oauth_last_used ON oauth_providers(last_used_at);

-- Sessions table for user authentication
CREATE TABLE sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    session_token VARCHAR(255) UNIQUE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE
);
-- Indexes for sessions table
CREATE INDEX idx_sessions_user_id ON sessions(user_id);
CREATE INDEX idx_sessions_session_token ON sessions(session_token);
CREATE INDEX idx_sessions_expires_at ON sessions(expires_at);
CREATE INDEX idx_sessions_user_expires ON sessions(user_id, expires_at);

-- Courses table to define available courses
CREATE TABLE courses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    year_id UUID NOT NULL,
    subject_id UUID NOT NULL,
    title VARCHAR(255) NOT NULL,
    description TEXT DEFAULT '',
    link_to_specification TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_by_user_id UUID,
    FOREIGN KEY (year_id) REFERENCES years(id) ON DELETE CASCADE,
    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE CASCADE,
    FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for courses table
CREATE INDEX idx_courses_year_id ON courses(year_id);
CREATE INDEX idx_courses_subject_id ON courses(subject_id);
CREATE INDEX idx_courses_year_subject ON courses(year_id, subject_id);
CREATE INDEX idx_courses_title ON courses(title);
CREATE INDEX idx_courses_is_active ON courses(is_active);
CREATE INDEX idx_courses_created_by ON courses(created_by_user_id);

-- Course enrollments table to track which users are enrolled in which courses
-- Users enroll at COURSE level, optionally specifying a tier for filtered content
CREATE TABLE course_enrollments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL,
    user_id UUID NOT NULL,
    tier_id UUID,  -- Optional: filter to specific tier (NULL = all tiers)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (tier_id) REFERENCES tiers(id) ON DELETE SET NULL,
    UNIQUE(course_id, user_id)
);
-- Indexes for course_enrollments table
CREATE INDEX idx_enrollments_course_id ON course_enrollments(course_id);
CREATE INDEX idx_enrollments_user_id ON course_enrollments(user_id);
CREATE INDEX idx_enrollments_user_course ON course_enrollments(user_id, course_id);
CREATE INDEX idx_enrollments_is_active ON course_enrollments(is_active);
CREATE INDEX idx_enrollments_tier_id ON course_enrollments(tier_id);
CREATE INDEX idx_enrollments_user_tier ON course_enrollments(user_id, tier_id);

-- Tiers table to define qualification levels within courses
-- Examples: Foundation/Higher (GCSE), AS/A2 (A-Level), Core/Extended (IGCSE)
CREATE TABLE tiers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL,
    title VARCHAR(255) NOT NULL,  -- e.g., "Foundation", "Higher", "AS Level", "A2 Level"
    code VARCHAR(50),              -- e.g., "FOUND", "HIGH", "AS", "A2", "FULL"
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for tiers table
CREATE INDEX idx_tiers_course_id ON tiers(course_id);
CREATE INDEX idx_tiers_is_active ON tiers(is_active);
CREATE INDEX idx_tiers_sort_order ON tiers(sort_order);
CREATE INDEX idx_tiers_added_by ON tiers(added_by_user_id);

-- Papers table to define assessments within courses
-- Papers now reference BOTH courses (required) and tiers (optional)
CREATE TABLE papers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL,            -- Direct course reference (required)
    tier_id UUID,                       -- Optional tier for grouping/filtering
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50),
    percentage_of_grade NUMERIC(5,2),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
    FOREIGN KEY (tier_id) REFERENCES tiers(id) ON DELETE SET NULL,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for papers table
CREATE INDEX idx_papers_course_id ON papers(course_id);
CREATE INDEX idx_papers_course_tier ON papers(course_id, tier_id);
CREATE INDEX idx_papers_tier_id ON papers(tier_id);
CREATE INDEX idx_papers_added_by ON papers(added_by_user_id);
CREATE INDEX idx_papers_code ON papers(code);
CREATE INDEX idx_papers_is_active ON papers(is_active);
CREATE UNIQUE INDEX idx_papers_unique_name_course_tier 
ON papers(course_id, COALESCE(tier_id, '00000000-0000-0000-0000-000000000000'::uuid), name) 
WHERE is_active = true;

-- Topics table to define subject areas within tiers
-- Topics belong to ONE tier (required) and are linked to papers via paper_topics junction table
-- If a topic appears in both Foundation and Higher, it must be duplicated (different content depth)
CREATE TABLE topics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tier_id UUID NOT NULL,             -- Tier this topic belongs to (required)
    course_id UUID NOT NULL,           -- Course reference (derived from tier, kept for convenience)
    paper_id UUID,                      -- DEPRECATED: Will be removed in future migration
    name VARCHAR(255) NOT NULL,
    start_page INTEGER,                 -- Start page (1-indexed, inclusive) in specification PDF
    end_page INTEGER,                   -- End page (1-indexed, inclusive) in specification PDF
    is_optional BOOLEAN DEFAULT FALSE NOT NULL,  -- Whether this topic is optional (user can choose to include/exclude)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (tier_id) REFERENCES tiers(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
    FOREIGN KEY (paper_id) REFERENCES papers(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for topics table
CREATE INDEX idx_topics_tier_id ON topics(tier_id);
CREATE INDEX idx_topics_course_id ON topics(course_id);
CREATE INDEX idx_topics_paper_id ON topics(paper_id);
CREATE INDEX idx_topics_added_by ON topics(added_by_user_id);
CREATE INDEX idx_topics_sort_order ON topics(sort_order);
CREATE INDEX idx_topics_is_active ON topics(is_active);
CREATE INDEX idx_topics_is_optional ON topics(is_optional);
-- Unique index to prevent duplicate topic names within same tier (case-insensitive)
CREATE UNIQUE INDEX idx_topics_tier_name_unique ON topics (tier_id, LOWER(name));
-- Index for topics with page ranges (useful for finding topics that have been processed)
CREATE INDEX idx_topics_has_page_range ON topics ((start_page IS NOT NULL AND end_page IS NOT NULL));

-- Paper-Topics junction table for many-to-many relationship
-- Allows topics to be linked to multiple papers within the same tier
CREATE TABLE paper_topics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    paper_id UUID NOT NULL,
    topic_id UUID NOT NULL,
    sort_order INTEGER DEFAULT 0,  -- Order of topic within this paper
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (paper_id) REFERENCES papers(id) ON DELETE CASCADE,
    FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE,
    UNIQUE(paper_id, topic_id)  -- Prevent duplicate links
);
-- Indexes for paper_topics table
CREATE INDEX idx_paper_topics_paper_id ON paper_topics(paper_id);
CREATE INDEX idx_paper_topics_topic_id ON paper_topics(topic_id);
CREATE INDEX idx_paper_topics_sort_order ON paper_topics(paper_id, sort_order);

-- User optional topics table for tracking which optional topics users have selected
-- Users can choose to include optional topics in their study plan
CREATE TABLE user_optional_topics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    topic_id UUID NOT NULL,
    course_enrollment_id UUID NOT NULL,  -- Links to specific course enrollment
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE,
    FOREIGN KEY (course_enrollment_id) REFERENCES course_enrollments(id) ON DELETE CASCADE,
    UNIQUE(user_id, topic_id)  -- User can only opt into a topic once
);
-- Indexes for user_optional_topics table
CREATE INDEX idx_user_optional_topics_user_id ON user_optional_topics(user_id);
CREATE INDEX idx_user_optional_topics_topic_id ON user_optional_topics(topic_id);
CREATE INDEX idx_user_optional_topics_enrollment ON user_optional_topics(course_enrollment_id);
CREATE INDEX idx_user_optional_topics_user_topic ON user_optional_topics(user_id, topic_id);

-- Notes table for text content within topics
CREATE TABLE notes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    topic_id UUID NOT NULL,
    content TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for notes table
CREATE INDEX idx_notes_topic_id ON notes(topic_id);
CREATE INDEX idx_notes_added_by ON notes(added_by_user_id);
CREATE INDEX idx_notes_is_active ON notes(is_active);

-- Questions table for practice questions within topics
CREATE TABLE questions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    topic_id UUID NOT NULL,
    title VARCHAR(500) DEFAULT '',
    sort_order INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for questions table
CREATE INDEX idx_questions_topic_id ON questions(topic_id);
CREATE INDEX idx_questions_added_by ON questions(added_by_user_id);
CREATE INDEX idx_questions_sort_order ON questions(sort_order);
CREATE INDEX idx_questions_is_active ON questions(is_active);

-- Question options table for multiple choice answers
CREATE TABLE question_options (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    question_id UUID NOT NULL,
    text TEXT NOT NULL,
    is_correct BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    FOREIGN KEY (question_id) REFERENCES questions(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for question_options table
CREATE INDEX idx_question_options_question_id ON question_options(question_id);
CREATE INDEX idx_question_options_added_by ON question_options(added_by_user_id);
CREATE INDEX idx_question_options_is_correct ON question_options(is_correct);

-- Past papers table for historical exam papers
CREATE TABLE past_papers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    paper_id UUID NOT NULL,
    year INTEGER NOT NULL,
    url TEXT NOT NULL,
    mark_scheme_url TEXT,
    file_size INTEGER,
    mark_scheme_file_size INTEGER,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (paper_id) REFERENCES papers(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for past_papers table
CREATE INDEX idx_past_papers_paper_id ON past_papers(paper_id);
CREATE INDEX idx_past_papers_added_by ON past_papers(added_by_user_id);
CREATE INDEX idx_past_papers_year ON past_papers(year);
CREATE INDEX idx_past_papers_is_active ON past_papers(is_active);
CREATE INDEX idx_past_papers_paper_year ON past_papers(paper_id, year);
CREATE INDEX idx_past_papers_mark_scheme_file_size ON past_papers(mark_scheme_file_size);

-- Podcasts table for audio content within topics
CREATE TABLE podcasts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    topic_id UUID NOT NULL,
    name VARCHAR(255) NOT NULL,
    script TEXT,  -- AI-generated podcast script (for TTS conversion)
    length_seconds INTEGER,
    file_size INTEGER,
    url TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    added_by_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for podcasts table
CREATE INDEX idx_podcasts_topic_id ON podcasts(topic_id);
CREATE INDEX idx_podcasts_added_by ON podcasts(added_by_user_id);
CREATE INDEX idx_podcasts_length ON podcasts(length_seconds);
CREATE INDEX idx_podcasts_is_active ON podcasts(is_active);
CREATE INDEX idx_podcasts_has_script ON podcasts ((script IS NOT NULL));

-- Uploaded files table for tracking R2 uploads
CREATE TABLE uploaded_files (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    object_key TEXT NOT NULL UNIQUE,
    file_url TEXT NOT NULL,
    original_filename TEXT NOT NULL,
    content_type TEXT,
    file_size BIGINT,
    uploaded_by_user_id UUID NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (uploaded_by_user_id) REFERENCES users(id) ON DELETE CASCADE
);
-- Indexes for uploaded_files table
CREATE INDEX idx_uploaded_files_object_key ON uploaded_files(object_key);
CREATE INDEX idx_uploaded_files_uploaded_by ON uploaded_files(uploaded_by_user_id);
CREATE INDEX idx_uploaded_files_created_at ON uploaded_files(created_at);

-- Models table for AI model configurations
CREATE TABLE models (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    model_name VARCHAR(255) NOT NULL UNIQUE,  -- Exact API model name (e.g., 'gemini-2.5-flash')
    provider VARCHAR(100) NOT NULL,  -- Provider name (e.g., 'google', 'cerebras', 'openai')
    rpd INTEGER,  -- Requests per day limit (NULL if unlimited)
    rpm INTEGER NOT NULL DEFAULT 10,  -- Requests per minute limit
    description TEXT,  -- Optional description of model capabilities
    is_active BOOLEAN DEFAULT TRUE,
    added_by_user_id UUID,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (added_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for models table
CREATE INDEX idx_models_provider ON models(provider);
CREATE INDEX idx_models_is_active ON models(is_active);
CREATE INDEX idx_models_model_name ON models(model_name);
CREATE INDEX idx_models_added_by ON models(added_by_user_id);

-- Generation prompts table for AI-powered content generation
-- Note: prompt_template uses Python .format() syntax with escaped braces
-- Double braces {{}} are literal, single braces {} are placeholders
CREATE TABLE generation_prompts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL UNIQUE,
    stage VARCHAR(50) NOT NULL,  -- 'course_info', 'papers_topics', 'notes', 'questions'
    prompt_template TEXT NOT NULL,
    model_id UUID NOT NULL,  -- References models table
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_by_user_id UUID,
    FOREIGN KEY (model_id) REFERENCES models(id) ON DELETE SET NULL,
    FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for generation_prompts table
CREATE INDEX idx_generation_prompts_stage ON generation_prompts(stage);
CREATE INDEX idx_generation_prompts_is_active ON generation_prompts(is_active);
CREATE INDEX idx_generation_prompts_name ON generation_prompts(name);
CREATE INDEX idx_generation_prompts_model_id ON generation_prompts(model_id);

-- Generation tasks table for tracking AI content generation jobs
CREATE TABLE generation_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    created_by_user_id UUID NOT NULL,
    specification_file_id UUID NOT NULL,
    specification_url TEXT NOT NULL,
    course_title VARCHAR(255),
    year_id UUID,
    year_name VARCHAR(50),
    subject_id UUID,
    subject_name VARCHAR(100),
    subject_code VARCHAR(10),
    create_new_subject BOOLEAN DEFAULT FALSE,
    course_description TEXT,
    course_id UUID,
    status VARCHAR(100) NOT NULL DEFAULT 'pending',
    total_papers INTEGER DEFAULT 0,
    total_topics INTEGER DEFAULT 0,
    completed_papers INTEGER DEFAULT 0,
    completed_topics INTEGER DEFAULT 0,
    completed_notes INTEGER DEFAULT 0,
    completed_podcast_scripts INTEGER DEFAULT 0,  -- Number of podcast scripts generated
    completed_podcast_audio INTEGER DEFAULT 0,    -- Number of podcast audio files generated
    completed_questions INTEGER DEFAULT 0,
    generate_podcasts BOOLEAN DEFAULT TRUE,       -- Whether to generate podcast scripts for this task
    generate_podcast_audio BOOLEAN DEFAULT FALSE, -- Whether to generate podcast audio using Inworld TTS
    speaker1_voice_id VARCHAR(100),               -- Inworld voice ID for Speaker 1
    speaker2_voice_id VARCHAR(100),               -- Inworld voice ID for Speaker 2
    speaker1_speed DECIMAL(3,2) DEFAULT 1.0,      -- Speaking rate for Speaker 1 (0.5 to 2.0)
    speaker2_speed DECIMAL(3,2) DEFAULT 1.0,      -- Speaking rate for Speaker 2 (0.5 to 2.0)
    pause_between_turns DECIMAL(4,2) DEFAULT 0.3, -- Pause duration in seconds between speaker turns
    error_message TEXT,
    error_details JSONB,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    started_at TIMESTAMP WITH TIME ZONE,
    info_completed_at TIMESTAMP WITH TIME ZONE,
    content_started_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,
    total_api_calls INTEGER DEFAULT 0,
    total_tokens_used INTEGER DEFAULT 0,
    estimated_cost DECIMAL(10, 4) DEFAULT 0.00,
    FOREIGN KEY (created_by_user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (specification_file_id) REFERENCES uploaded_files(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE SET NULL,
    FOREIGN KEY (year_id) REFERENCES years(id) ON DELETE SET NULL,
    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE SET NULL
);
-- Indexes for generation_tasks table
CREATE INDEX idx_generation_tasks_status ON generation_tasks(status);
CREATE INDEX idx_generation_tasks_created_by ON generation_tasks(created_by_user_id);
CREATE INDEX idx_generation_tasks_course_id ON generation_tasks(course_id);
CREATE INDEX idx_generation_tasks_created_at ON generation_tasks(created_at DESC);
CREATE INDEX idx_generation_tasks_specification ON generation_tasks(specification_file_id);
CREATE INDEX idx_generation_tasks_audio ON generation_tasks(generate_podcast_audio);

-- User question performance table for tracking user analytics on questions
CREATE TABLE user_question_performance (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    question_id UUID NOT NULL,
    times_seen INTEGER DEFAULT 0 NOT NULL,
    times_correct INTEGER DEFAULT 0 NOT NULL,
    times_incorrect INTEGER DEFAULT 0 NOT NULL,
    ema_accuracy DECIMAL(5,4) DEFAULT 0.88 NOT NULL,  -- EMA accuracy (0.0 to 1.0)
    total_time_spent_ms BIGINT DEFAULT 0 NOT NULL,    -- Cumulative time spent on this question (milliseconds)
    recent_results INTEGER DEFAULT NULL,              -- Recent-N bitfield: last 16 attempts. Bit 0 = most recent. 1=correct, 0=incorrect. NULL = no data (pre-migration rows).
    last_seen_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (question_id) REFERENCES questions(id) ON DELETE CASCADE,
    UNIQUE(user_id, question_id)
);
-- Indexes for user_question_performance table
CREATE INDEX idx_user_question_perf_user_id ON user_question_performance(user_id);
CREATE INDEX idx_user_question_perf_question_id ON user_question_performance(question_id);
CREATE INDEX idx_user_question_perf_user_question ON user_question_performance(user_id, question_id);
CREATE INDEX idx_user_question_perf_last_seen ON user_question_performance(last_seen_at DESC);
CREATE INDEX idx_user_question_perf_ema_accuracy ON user_question_performance(user_id, ema_accuracy);
CREATE INDEX idx_user_question_perf_time_spent ON user_question_performance(user_id, total_time_spent_ms);

-- User daily XP table for tracking daily progress and weekly leaderboards
CREATE TABLE user_daily_xp (
    user_id UUID NOT NULL,
    date DATE NOT NULL,
    xp_earned INTEGER DEFAULT 0 NOT NULL,
    questions_answered INTEGER DEFAULT 0 NOT NULL,
    questions_correct INTEGER DEFAULT 0 NOT NULL,     -- Correct answers per day (for accuracy)
    time_spent_seconds INTEGER DEFAULT 0 NOT NULL,    -- Daily study time in seconds
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, date),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
-- Indexes for user_daily_xp table
CREATE INDEX idx_daily_xp_date ON user_daily_xp(date DESC);
CREATE INDEX idx_daily_xp_user_date ON user_daily_xp(user_id, date DESC);
CREATE INDEX idx_daily_xp_questions ON user_daily_xp(user_id, questions_answered, questions_correct);

-- Streak freeze usage table - tracks when freezes were applied to protect streaks
-- Users get 14 freezes per calendar year. When a user misses a day, a freeze is
-- automatically applied to preserve their streak (if freezes are available).
CREATE TABLE streak_freeze_usage (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    freeze_date DATE NOT NULL,             -- The date that was "frozen" (the missed day)
    was_auto_applied BOOLEAN DEFAULT TRUE NOT NULL,  -- TRUE if auto-applied, FALSE if manually scheduled
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE(user_id, freeze_date)           -- User can only have one freeze per date
);
-- Indexes for streak_freeze_usage table
CREATE INDEX idx_streak_freeze_user_id ON streak_freeze_usage(user_id);
CREATE INDEX idx_streak_freeze_date ON streak_freeze_usage(freeze_date);
CREATE INDEX idx_streak_freeze_user_date ON streak_freeze_usage(user_id, freeze_date DESC);

-- Subscription tiers table - stores available subscription plans
CREATE TABLE subscription_tiers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL UNIQUE,     -- 'Lockin Free', 'Lockin Pro', 'Lockin Super'
    code VARCHAR(50) NOT NULL UNIQUE,      -- 'free', 'pro', 'super'
    price_gbp DECIMAL(10,2) NOT NULL DEFAULT 0.00,  -- One-time price in GBP
    description TEXT,
    
    -- Feature limits (NULL = unlimited)
    questions_per_day INTEGER,             -- NULL for unlimited, 50 for Free
    podcasts_percentage INTEGER,           -- Percentage of topics (50 for Free, 100 for Pro/Super)
    
    -- Feature flags
    full_analytics BOOLEAN DEFAULT FALSE,      -- Full analytics access
    full_xp_earning BOOLEAN DEFAULT FALSE,     -- Full XP multiplier
    full_leaderboards BOOLEAN DEFAULT FALSE,   -- Full leaderboard participation
    
    -- Metadata
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
-- Indexes for subscription_tiers table
CREATE INDEX idx_subscription_tiers_code ON subscription_tiers(code);
CREATE INDEX idx_subscription_tiers_is_active ON subscription_tiers(is_active);
CREATE INDEX idx_subscription_tiers_sort_order ON subscription_tiers(sort_order);

-- User subscriptions table - links users to their subscription tiers
CREATE TABLE user_subscriptions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    subscription_tier_id UUID NOT NULL,
    
    -- Purchase tracking
    purchased_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    purchased_price_gbp DECIMAL(10,2),     -- Price at time of purchase (historical)
    payment_method VARCHAR(100),           -- 'auto_assigned', 'admin_grant', 'stripe', 'paypal', etc.
    transaction_id VARCHAR(255),           -- External payment reference
    
    -- Status tracking
    is_active BOOLEAN DEFAULT TRUE,
    deactivated_at TIMESTAMP WITH TIME ZONE,
    deactivated_by_user_id UUID,           -- Admin who deactivated (if applicable)
    deactivation_reason TEXT,
    
    -- Metadata
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (subscription_tier_id) REFERENCES subscription_tiers(id) ON DELETE RESTRICT,
    FOREIGN KEY (deactivated_by_user_id) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes for user_subscriptions table
CREATE INDEX idx_user_subscriptions_user_id ON user_subscriptions(user_id);
CREATE INDEX idx_user_subscriptions_tier_id ON user_subscriptions(subscription_tier_id);
CREATE INDEX idx_user_subscriptions_is_active ON user_subscriptions(is_active);
CREATE INDEX idx_user_subscriptions_user_active ON user_subscriptions(user_id, is_active);
CREATE INDEX idx_user_subscriptions_purchased_at ON user_subscriptions(purchased_at DESC);
-- Unique constraint: only one active subscription per user
CREATE UNIQUE INDEX idx_user_subscriptions_one_active ON user_subscriptions(user_id) WHERE is_active = true;

-- Subscription Quota Tracking:
-- Daily question quota is tracked in the existing user_daily_xp.questions_answered column
-- Podcast access is determined by deterministic hash: (MD5(topic_id) % 100) < podcasts_percentage

-- Study sessions table to track practice sessions (regular and burst)
CREATE TABLE study_sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    course_id UUID,                                          -- NULL if cross-course or unscoped
    session_type VARCHAR(20) NOT NULL DEFAULT 'regular',     -- 'regular' or 'burst'
    target_duration_seconds INTEGER,                         -- NULL for regular sessions, set for burst
    actual_duration_seconds INTEGER DEFAULT 0,
    xp_earned INTEGER DEFAULT 0,
    questions_answered INTEGER DEFAULT 0,
    questions_correct INTEGER DEFAULT 0,
    status VARCHAR(20) NOT NULL DEFAULT 'active',            -- 'active', 'completed', 'abandoned'
    started_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE SET NULL
);
-- Indexes for study_sessions table
CREATE INDEX idx_study_sessions_user_id ON study_sessions(user_id);
CREATE INDEX idx_study_sessions_user_status ON study_sessions(user_id, status);
CREATE INDEX idx_study_sessions_user_type ON study_sessions(user_id, session_type);
CREATE INDEX idx_study_sessions_course_id ON study_sessions(course_id);
CREATE INDEX idx_study_sessions_started_at ON study_sessions(started_at DESC);
CREATE INDEX idx_study_sessions_type ON study_sessions(session_type);
CREATE INDEX idx_study_sessions_status ON study_sessions(status);
